{{/* Site-specific CSS and JS. */}} {{- /* Dark only. The theme reads `wc-color-theme` from localStorage before falling back to the configured default, so a browser that toggled to light once stays light forever now that the toggle is hidden (params.yaml). This normalises the stored value and paints the class immediately. INLINE and BEFORE the stylesheet on purpose: as a deferred file it would run after first paint and every visit would flash white. It is the one script that has to be inline — everything else goes through the fingerprinted bundle above. */ -}} {{- /* `hide_title: true` in a page's front matter drops the rendered page title, for pages that open on their own first heading instead (/research/). The title is still set, so the nav, the tab, Pagefind and the sitemap all keep it. The theme's single.html has no hook and no parameter for this, hence the rule here rather than a forked layout. */ -}} {{- with .Params.hide_title -}} {{- end -}} {{- $css := slice -}} {{- range (resources.Match "css/custom/*.css") -}} {{- $css = $css | append . -}} {{- end -}} {{- /* STABLE filenames, no fingerprint, no SRI -- deliberately. These were fingerprinted (`custom-bundle.min..css`) with an integrity attribute. Each deploy replaced the file, and GitHub Pages serves HTML with `Cache-Control: max-age=600`, so for up to ten minutes after every deploy a returning visitor's cached page asked for a stylesheet that no longer existed: 404, and the site rendered as the raw theme -- centred titles, stacked tiles, the education block that 03-biography.css hides. The JS failed the same way, silently. With a stable name, the worst case for those ten minutes is the PREVIOUS stylesheet against the new HTML: mildly stale, never unstyled. Pages caps assets at the same ten minutes, so there is no long-lived staleness to bust; and SRI on a same-origin file adds nothing here while guaranteeing the failure above whenever the hash in cached HTML stops matching. */ -}} {{- with $css -}} {{- $bundle := . | resources.Concat "css/custom-bundle.css" | minify -}} {{- else -}} {{- errorf "custom-assets.html: no stylesheets matched assets/css/custom/*.css" -}} {{- end -}} {{- range slice "js/biography-aria.js" "js/research-strip.js" "js/figure-lightbox.js" "js/toc-scrollspy.js" -}} {{- with resources.Get . -}} {{- $js := . | minify -}} {{- else -}} {{- errorf "custom-assets.html: missing asset %q" . -}} {{- end -}} {{- end -}}
